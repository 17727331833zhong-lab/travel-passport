        var RESOURCE_KEYS = ['template0', 'template1', 'template2', 'template3'];


        function getResource(key) { try { return localStorage.getItem('res_' + key); } catch (e) { return null; } }


        function setResource(key, data) { try { localStorage.setItem('res_' + key, data); } catch (e) {} }


        function uploadResource(key, input) {

            var file = input.files[0];

            if (!file) return;

            var reader = new FileReader();

            reader.onload = function(e) {

                var dataUrl = e.target.result;

                setResource(key, dataUrl);

                document.getElementById('uploadStatus').innerText = '✅ ' + key + ' 上传成功！';

                applyResource(key, dataUrl);

                var previewEl = document.getElementById(key + 'Preview');

                if (previewEl) previewEl.innerHTML = '<img src="' + dataUrl + '" />';

                showToast(key + ' 上传成功 ✅', 'success');

            };

            reader.readAsDataURL(file);

        }


        function applyResource(key, dataUrl) {

            if (key.startsWith('template')) {

                var idx = parseInt(key.replace('template', ''));

                if (!isNaN(idx)) { window._templateImages = window._templateImages || {};

                    window._templateImages[idx] = dataUrl; }

            }

        }


        function loadAllResources() {

            RESOURCE_KEYS.forEach(function(key) {

                var data = getResource(key);

                if (data) {

                    applyResource(key, data);

                    var previewEl = document.getElementById(key + 'Preview');

                    if (previewEl) previewEl.innerHTML = '<img src="' + data + '" />';

                }

            });

            for (var i = 0; i < 4; i++) {

                var data = getResource('template' + i);

                if (data) { window._templateImages = window._templateImages || {};

                    window._templateImages[i] = data; }

            }

        }


        function clearResources() {

            if (!confirm('确定清除所有上传资源？')) return;

            showToast('已清除所有资源', '');

            setTimeout(function() { location.reload(); }, 500);

        }

        function openCamera() {

            var captureBtn = document.getElementById('camCaptureBtn');

                showToast('浏览器不支持摄像头', 'error');

                    video.style.display = 'none';

                    canvas.style.display = 'block';

                    openBtn.style.display = 'block';

                    captureBtn.style.display = 'none';

                    retryBtn.style.display = 'none';

                });

        }
